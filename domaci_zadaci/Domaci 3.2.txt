--3.2.2 Prikažite ime i prezime igrača i naziv kluba za koji je igrao u sezoni 2024/25. Uklonite duplikate iz rezultata.
SELECT DISTINCT
  p.first_name,
  p.last_name,
  t.team_name AS klub
FROM player_match_stats s
JOIN matches m
  ON m.match_id = s.match_id
JOIN players p
  ON p.player_id = s.player_id
JOIN teams t
  ON t.team_id = s.team_id
WHERE m.season = '2024/25'
ORDER BY p.last_name, p.first_name, klub;

-- 3.2.3 Prikažite datum odigravanja utakmice, naziv stadiona i broj gledalaca za sve utakmice u sezoni 2024/25.
SELECT
  CAST(m.kickoff_ts AS DATE) AS datum,
  s.stadium_name AS stadion,
  m.attendance AS broj_gledalaca
FROM matches m
JOIN stadiums s
  ON s.stadium_id = m.stadium_id
WHERE m.season = '2024/25'
ORDER BY m.kickoff_ts;

--3.2.4. 1.Prikažite sve utakmice za koje nije evidentiran broj gledalaca. Prikažite datum, domaći klub i gostujući klub.
SELECT
  CAST(m.kickoff_ts AS DATE) AS datum,
  th.team_name AS domaci_klub,
  ta.team_name AS gostujuci_klub
FROM matches m
JOIN teams th
  ON th.team_id = m.home_team_id
JOIN teams ta
  ON ta.team_id = m.away_team_id
WHERE m.attendance IS NULL
ORDER BY m.kickoff_ts;

--3.2.5 Pronađite stadione na kojima nije odigrana ni jedna utakmica u sezoni 2024/25.
SELECT
  s.stadium_id,
  s.stadium_name,
  s.city,
  s.capacity
FROM stadiums s
LEFT JOIN matches m
  ON m.stadium_id = s.stadium_id
 AND m.season = '2024/25'
WHERE m.match_id IS NULL;

--3.2.6 Za svaki klub izračunajte ukupan broj žutih kartona koje su dobili njegovi igrači u sezoni 2024/25. Rezultat sortirajte opadajuće prema broju kartona.
SELECT
  t.team_name AS klub,
  COUNT(*) AS zuti_kartoni
FROM match_events e
JOIN matches m
  ON m.match_id = e.match_id
JOIN teams t
  ON t.team_id = e.team_id
WHERE m.season = '2024/25'
  AND e.event_type = 'yellow_card'
GROUP BY t.team_id, t.team_name
ORDER BY zuti_kartoni DESC, klub;

--3.2.7 Pronađite pet igrača sa najvećim brojem asistencija u sezoni 2024/25. Prikažite ime igrača i broj asistencija.
SELECT
  CONCAT(p.first_name, ' ', p.last_name) AS igrac,
  COUNT(*) AS asistencije
FROM match_events e
JOIN matches m
  ON m.match_id = e.match_id
JOIN players p
  ON p.player_id = e.related_player_id
WHERE m.season = '2024/25'
  AND e.event_type = 'goal'
GROUP BY p.player_id, p.first_name, p.last_name
ORDER BY asistencije DESC, p.last_name, p.first_name
LIMIT 5;

--3.2.8 2.Pronađite tri igrača sa najvećim ukupnim brojem odigranih minuta u sezoni 2024/25. Prikažite ime igrača, klub i ukupan broj minuta.
SELECT
  CONCAT(p.first_name, ' ', p.last_name) AS igrac,
  ARRAY_JOIN(COLLECT_SET(t.team_name), ', ') AS klub,
  SUM(s.minutes_played) AS ukupno_minuta
FROM player_match_stats s
JOIN matches m
  ON m.match_id = s.match_id
JOIN players p
  ON p.player_id = s.player_id
JOIN teams t
  ON t.team_id = s.team_id
WHERE m.season = '2024/25'
GROUP BY p.player_id, p.first_name, p.last_name
ORDER BY ukupno_minuta DESC, igrac
LIMIT 3;

--3.2.9 3.Za svaku sezonu prikažite broj isključenja po klubovima. 
SELECT
  m.season,
  t.team_name AS klub,
  COUNT(*) AS iskljucenja
FROM match_events e
JOIN matches m
  ON m.match_id = e.match_id
JOIN teams t
  ON t.team_id = e.team_id
WHERE e.event_type IN ('red_card', 'second_yellow')
GROUP BY m.season, t.team_id, t.team_name
ORDER BY m.season, iskljucenja DESC, klub;
